* Computer-specific settings

Load some computer-specific settings, such as the name and email address.

These are keyed by short hostname, not by login name: =users/= holds one
subdirectory per machine, and only the one matching =hostname -s= is loaded.
Logging in as a different user on the same machine loads the same files.

#+begin_src emacs-lisp
(use-package emacs
  :ensure nil
  :init
  (defvar nl/host-settings-dir nil
    "Directory of Emacs settings specific to this machine.
Named after the short hostname, under `users'.")

  (require 's)
  (let ((hosts-dir (concat user-emacs-directory "users")))
    (when (file-directory-p hosts-dir)
      (setq nl/host-settings-dir
            (concat hosts-dir "/" (s-trim (shell-command-to-string "hostname -s"))))
      (add-to-list 'load-path nl/host-settings-dir)))

  ;; Load the settings for this host, if any exist.
  (when (and nl/host-settings-dir (file-exists-p nl/host-settings-dir))
    (mapc 'load (directory-files nl/host-settings-dir nil "^[^#].*el$"))))
#+end_src
