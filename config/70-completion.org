                                (executable-find "bat")
                                yuta-bat-executable))

    "Return a bat preview command for FILE, preserving decorations and color."
    (if (and yuta-bat-executable (executable-find yuta-bat-executable))
        (concat (shell-quote-argument yuta-bat-executable)
                " --color=always --decorations=always --style=numbers,header"
                " --line-range "
                (if line
                    (format "%d:%d" (max 1 (- line 10))
                            (+ line yuta-preview-lines))
                  (format "1:%d" yuta-preview-lines))
                (when line (format " --highlight-line %d" line))
                " -- " (shell-quote-argument file))
      (if line
          (format "sed -n '%d,%dp' %s"
                  (max 1 (- line 5))
                  (+ line yuta-preview-lines)
                  (shell-quote-argument file))
        (format "head -n %d %s"
                yuta-preview-lines
                (shell-quote-argument file)))))

                                  "--exclude public --exclude database --exclude e2e "
                                  ". %s")
                          yuta-fd-executable
                          (shell-quote-argument root)))))
      (yuta--launch 'files cmd)))

(defun nl/consult-ripgrep-project-paths (paths)
  "Run `consult-ripgrep` in the Projectile project's PATHS.
PATHS are relative to the project root."
  (require 'consult)
  (let ((project-root (projectile-project-root)))
    (if (not project-root)
        (message "Not in a Projectile project.")
      (let ((existing-paths
             (delq nil
                   (mapcar (lambda (path)
                             (let ((expanded-path
                                    (expand-file-name path project-root)))
                               (when (file-directory-p expanded-path)
                                 expanded-path)))
                           paths))))
        (if (null existing-paths)
            (message "None of these folders exist in project: %s"
                     (mapconcat #'identity paths ", "))
          (consult-ripgrep existing-paths))))))
