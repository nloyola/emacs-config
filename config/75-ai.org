(use-package chatgpt-shell
  :ensure t
  :custom
  ((chatgpt-shell-openai-key
    (lambda ()
      (auth-source-pick-first-password :host "api.openai.com"))))
  :config
  (setq chatgpt-shell-model-version "chatgpt-4o-latest")
  )
