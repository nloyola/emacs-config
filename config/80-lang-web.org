(defun nl/tsx-comment-dwim (arg)
  "Toggle JSX comment {/* */} on each line in region or current line."
  (interactive "P")
  (save-excursion
    (let* ((use-region (use-region-p))
           (beg (if use-region
                    (region-beginning)
                  (line-beginning-position)))
           (end (if use-region
                    (region-end)
                  (line-end-position)))
           (end-line (progn
                       (goto-char end)
                       (line-number-at-pos))))
      (goto-char beg)
      (let ((commented (looking-at "[ \t]*{/\\*")))
        (while (<= (line-number-at-pos) end-line)
          (beginning-of-line)
          (if commented
              (progn
                (when (looking-at "[ \t]*{/\\* ?")
                  (delete-region
                   (match-beginning 0)
                   (+ (match-beginning 0) 4)))
                (when (search-forward " */}" (line-end-position) t)
                  (delete-region (- (point) 4) (point))))
            (insert "{/* ")
            (end-of-line)
            (insert " */}"))
          (forward-line 1))))))

  (defun nl/web-mode-text-node-p ()
    "Return non-nil when point is in plain markup text."
    (and (not (nth 8 (syntax-ppss)))
         (not (get-text-property (point) 'tag-type))
         (not (get-text-property (point) 'block-side))
         (not (get-text-property (point) 'part-side))

    "Fill text/comments/strings or reindent the current web-mode block/element."
    (interactive "P")
    (save-excursion
      (cond
       ((or (nth 8 (syntax-ppss))
            (nl/web-mode-text-node-p))
        (let ((fill-prefix (save-excursion
                             (back-to-indentation)

    "Return the indentation of the matching standalone PHP if line."
    (save-excursion
      (let ((depth 1))
        (forward-line 0)
        (while (and (> depth 0)
                    (re-search-backward
                     "^[[:space:]]*<\\?php[[:space:]]+\\(if\\|endif\\)\\b" nil t))
          (if (string= (match-string 1) "endif")

  (defun nl/web-mode-indent-region (start end)
    "Indent region line-by-line with PHP template control awareness."
    (save-excursion
      (let ((end-marker (copy-marker end))
            (offset web-mode-code-indent-offset)
            stack)
        (goto-char (point-min))
        (while (< (point) end-marker)
          (let ((in-region (>= (point) start))
                (kind (nl/web-mode-php-control-kind)))
            (unless (looking-at-p "[[:space:]]*$")
              (cond
               ((member kind '("else" "elseif" "endif"))
                (when in-region
                  (web-mode-indent-line)
                  (when stack
                    (indent-line-to (car stack))))
                (when (string= kind "endif")
                  (pop stack)))
               (t
                (when in-region
                  (web-mode-indent-line)
                  (when (and stack (< (current-indentation) (+ (car stack) offset)))
                    (indent-line-to (+ (car stack) offset)))))))
            (when (string= kind "if")
              (push (current-indentation) stack)))
          (forward-line 1))
        (set-marker end-marker nil))))

    "Fill text inside YAML block scalars without swallowing YAML structure."
    (interactive "P")
    (let ((indent (current-indentation)))
      (if (and (> indent 0)
               (nl/yaml--block-scalar-header-before-p indent))
          (let* ((prefix (make-string indent ?\s))

                          (while (and (= (forward-line -1) 0)
                                      (not (looking-at-p "^[[:space:]]*$"))

** [[https://polymode.github.io/][polymode]] for YAML with embedded HTML

Treat the block-scalar value (~|~, ~|-~, ~>~, …) of selected keys as
HTML.  The region starts at the line after ~key: |-~ and ends at the
first line whose indentation is less than or equal to the key's
indent (or end of buffer).  Add keys to ~nl/poly-yaml-html-keys~ to
extend.

#+begin_src emacs-lisp
(use-package polymode
  :mode ("\\.\\(yml\\|yaml\\)\\'" . nl/poly-yaml-mode)
  :init (require 'yaml-mode nil t)
  :preface
  (defvar nl/poly-yaml-html-keys
    '("body" "content" "description" "html")
    "YAML keys whose block-scalar values are edited as HTML.")

  (defun nl/poly-yaml-html--head-re ()
    (concat "^\\([ \t]*\\)\\(?:"
            (mapconcat #'regexp-quote nl/poly-yaml-html-keys "\\|")
            "\\):[ \t]*[|>][-+]?[ \t]*\n"))

  (defun nl/poly-yaml-html-head-matcher (ahead)
    "Find a YAML block-scalar header for an HTML key."
    (when (if (< ahead 0)
              (re-search-backward (nl/poly-yaml-html--head-re) nil t (- ahead))
            (re-search-forward (nl/poly-yaml-html--head-re) nil t ahead))
      (cons (match-beginning 0) (match-end 0))))

  (defun nl/poly-yaml-html-tail-matcher (_ahead)
    "Find the end of the block scalar opened by the preceding head."
    (let ((indent
           (save-excursion
             (when (re-search-backward (nl/poly-yaml-html--head-re) nil t)
               (length (match-string 1))))))
      (if (not indent)
          (cons (point-max) (point-max))
        (let ((re (format "^[ \t]\\{0,%d\\}[^ \t\n]" indent)))
          (if (re-search-forward re nil t)
              (cons (match-beginning 0) (match-beginning 0))
            (cons (point-max) (point-max)))))))
  (defun nl/poly-yaml-inner-fill ()
    "Match inner-chunk fill settings to the YAML host."
    (setq fill-column 120)
    ;; Inner chunks use mhtml-mode for YAML body/content HTML.  Keep M-q
    ;; YAML-aware so filling cannot escape the current block scalar and
    ;; collapse surrounding YAML structure.
    (setq-local fill-paragraph-function #'nl/yaml-fill-paragraph)
    (nl/yaml-enable-local-keys)
    (auto-fill-mode 1))
  :hook
  (polymode-init-inner-hook . nl/poly-yaml-inner-fill)
  (nl/poly-yaml-mode . nl/poly-yaml-inner-fill)
  :config
  (define-hostmode nl/poly-yaml-hostmode
    :mode 'yaml-mode)
  (define-innermode nl/poly-yaml-html-innermode
    :mode 'mhtml-mode
    :head-matcher #'nl/poly-yaml-html-head-matcher
    :tail-matcher #'nl/poly-yaml-html-tail-matcher
    :head-mode 'host
    :tail-mode 'host)
  (define-polymode nl/poly-yaml-mode
    :hostmode 'nl/poly-yaml-hostmode
    :innermodes '(nl/poly-yaml-html-innermode)))
