(defun nl/tsx-comment-dwim (arg)
  "Toggle JSX comment {/* */} on each line in region or current line."
  (interactive "P")
  (save-excursion
    (let* ((use-region (use-region-p))
           (beg (if use-region
                    (region-beginning)
                  (line-beginning-position)))
           (end (if use-region
                    (region-end)
                  (line-end-position)))
           (end-line (progn
                       (goto-char end)
                       (line-number-at-pos))))
      (goto-char beg)
      (let ((commented (looking-at "[ \t]*{/\\*")))
        (while (<= (line-number-at-pos) end-line)
          (beginning-of-line)
          (if commented
              (progn
                (when (looking-at "[ \t]*{/\\* ?")
                  (delete-region
                   (match-beginning 0)
                   (+ (match-beginning 0) 4)))
                (when (search-forward " */}" (line-end-position) t)
                  (delete-region (- (point) 4) (point))))
            (insert "{/* ")
            (end-of-line)
            (insert " */}"))
          (forward-line 1))))))

(use-package polymode
  :mode ("\\.\\(yml\\|yaml\\)\\'" . nl/poly-yaml-mode)
  :init (require 'yaml-mode nil t)
  :preface
  (defvar nl/poly-yaml-html-keys
    '("body" "content" "description" "html")
    "YAML keys whose block-scalar values are edited as HTML.")

  (defun nl/poly-yaml-html--head-re ()
    (concat "^\\([ \t]*\\)\\(?:"
            (mapconcat #'regexp-quote nl/poly-yaml-html-keys "\\|")
            "\\):[ \t]*[|>][-+]?[ \t]*\n"))

  (defun nl/poly-yaml-html-head-matcher (ahead)
    "Find a YAML block-scalar header for an HTML key."
    (when (if (< ahead 0)
              (re-search-backward (nl/poly-yaml-html--head-re) nil t (- ahead))
            (re-search-forward (nl/poly-yaml-html--head-re) nil t ahead))
      (cons (match-beginning 0) (match-end 0))))

  (defun nl/poly-yaml-html-tail-matcher (_ahead)
    "Find the end of the block scalar opened by the preceding head."
    (let ((indent
           (save-excursion
             (when (re-search-backward (nl/poly-yaml-html--head-re) nil t)
               (length (match-string 1))))))
      (if (not indent)
          (cons (point-max) (point-max))
        (let ((re (format "^[ \t]\\{0,%d\\}[^ \t\n]" indent)))
          (if (re-search-forward re nil t)
              (cons (match-beginning 0) (match-beginning 0))
            (cons (point-max) (point-max)))))))
  (defun nl/poly-yaml-inner-fill ()
    "Match inner-chunk fill settings to the YAML host."
    (setq fill-column 120)
    ;; Inner chunks use mhtml-mode for YAML body/content HTML.  Keep M-q
    ;; YAML-aware so filling cannot escape the current block scalar and
    ;; collapse surrounding YAML structure.
    (setq-local fill-paragraph-function #'nl/yaml-fill-paragraph)
    (nl/yaml-enable-local-keys)
    (auto-fill-mode 1))
  :hook
  (polymode-init-inner-hook . nl/poly-yaml-inner-fill)
  (nl/poly-yaml-mode . nl/poly-yaml-inner-fill)
  :config
  (define-hostmode nl/poly-yaml-hostmode
    :mode 'yaml-mode)
  (define-innermode nl/poly-yaml-html-innermode
    :mode 'mhtml-mode
    :head-matcher #'nl/poly-yaml-html-head-matcher
    :tail-matcher #'nl/poly-yaml-html-tail-matcher
    :head-mode 'host
    :tail-mode 'host)
  (define-polymode nl/poly-yaml-mode
    :hostmode 'nl/poly-yaml-hostmode
    :innermodes '(nl/poly-yaml-html-innermode)))
