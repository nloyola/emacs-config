** [[https://github.com/magit/magit][magit]]

A great interface for git projects. It's much more pleasant to use than the git interface on the
command line. Use an easy keybinding to access magit.

#+begin_src emacs-lisp
(use-package magit
  ;;:pin melpa
  :bind (("C-x g" . magit-status))
  ;;:hook (magit-mode . magit-todos-mode)
  :preface
  (defun nl/git-commit-load-maybe ()
    "Load and activate `git-commit' for Git commit message files."
    (when (and buffer-file-name
               (string-match-p "/\\(\\(\\(COMMIT\\|NOTES\\|PULLREQ\\|MERGEREQ\\|TAG\\)_EDIT\\|MERGE_\\|\\)MSG\\|\\(BRANCH\\|EDIT\\)_DESCRIPTION\\)\\'"
                               buffer-file-name))
      (require 'git-commit)
      (git-commit-setup-check-buffer)))
  :init
  (add-hook 'find-file-hook #'nl/git-commit-load-maybe)
  :config
  (defun nl/git-commit-fill-paragraph (&optional justify region)
    "Fill bullet lists normally in commit messages.

`git-commit-setup-changelog-support' installs `log-edit-fill-entry',
which treats leading `*' lines as ChangeLog entries.  That is useful
for actual ChangeLog-style paragraphs, but it makes ordinary bullet
lists wrap under the bullet instead of under the text."
    (save-excursion
      (back-to-indentation)
      (if (looking-at "\\(?:[-+*]\\|[0-9]+[.)]\\)[[:blank:]]+")
          (let* ((text-col (save-excursion
                             (goto-char (match-end 0))
                             (current-column)))
                 (fill-prefix (make-string text-col ?\s))
                 (adaptive-fill-mode nil)
                 (start (line-beginning-position))
                 (end (copy-marker
                       (save-excursion
                         (forward-line 1)
                         (while (and (not (eobp))
                                     (not (looking-at
                                           "[ \t]*\\(?:$\\|\\(?:[-+*]\\|[0-9]+[.)]\\)[ \t]+\\)")))
                           (forward-line 1))
                         (point)))))
            ;; Strip continuation-line indentation so `fill-region-as-paragraph'
            ;; can re-wrap cleanly against our explicit fill-prefix.
            (save-excursion
              (goto-char start)
              (forward-line 1)
              (while (< (point) end)
                (delete-horizontal-space)
                (forward-line 1)))
            (fill-region-as-paragraph start end justify))
        (log-edit-fill-entry justify)))
    t)

  (defun nl/git-commit-cleanup-ai-message ()
    "Unindent and refill the commit body after the summary line.

This is intended for AI-generated commit messages where each body
line is indented uniformly.  The summary line is left untouched,
all non-blank body lines are shifted left by their shared leading
indentation, and each resulting paragraph is refilled."
    (interactive)
    (let* ((body-start (save-excursion
                         (goto-char (point-min))
                         (forward-line 1)
                         (point)))
           (body-end (copy-marker
                      (save-excursion
                        (goto-char (point-min))
                        (if (re-search-forward (concat "^" (regexp-quote comment-start)) nil t)
                            (line-beginning-position)
                          (point-max)))))
           (min-indent nil))
      (save-excursion
        (goto-char body-start)
        (while (< (point) body-end)
          (unless (looking-at-p "[ \t]*$")
            (back-to-indentation)
            (setq min-indent
                  (if min-indent
                      (min min-indent (current-column))
                    (current-column))))
          (forward-line 1))
        (when (and min-indent (> min-indent 0))
          (indent-rigidly body-start body-end (- min-indent)))
        (save-restriction
          (narrow-to-region body-start body-end)
          (goto-char (point-min))
          (while (< (point) (point-max))
            (cond
             ((looking-at-p "[ \t]*$")
              (forward-line 1))
             ((looking-at "[ \t]*\\(?:[-+*]\\|[0-9]+[.)]\\)[ \t]+")
              (nl/git-commit-fill-paragraph)
              (forward-line 1)
              (while (and (not (eobp))
                          (not (looking-at-p
                                "[ \t]*\\(?:$\\|\\(?:[-+*]\\|[0-9]+[.)]\\)[ \t]+\\)")))
                (forward-line 1)))
             (t
              (let ((start (point))
                    (end (save-excursion
                           (forward-line 1)
                           (while (and (not (eobp))
                                       (not (looking-at-p
                                             "[ \t]*\\(?:$\\|\\(?:[-+*]\\|[0-9]+[.)]\\)[ \t]+\\)")))
                             (forward-line 1))
                           (point))))
                (fill-region-as-paragraph start end)
                (goto-char end)))))))
      (set-marker body-end nil)
      (message "Cleaned up commit message body")))

  (defun nl/git-commit-setup-fill-paragraph ()
    "Use better paragraph filling for bullet lists in commit buffers."
    (setq-local fill-paragraph-function #'nl/git-commit-fill-paragraph)
    (local-set-key (kbd "C-c M-q") #'nl/git-commit-cleanup-ai-message)
    ;; Treat each bullet line as the start of its own paragraph, so
    ;; `M-q' on a bullet only fills that item even without blank lines
    ;; between bullets.
    (setq-local paragraph-start
                "\f\\|[ \t]*$\\|[ \t]*\\(?:[-+*]\\|[0-9]+[.)]\\)[ \t]+")
    (setq-local paragraph-separate "\f\\|[ \t]*$"))

  (defun nl/git-commit-empty-message-on-cancel ()
    "Make a canceled commit safe even if the editor client exits successfully."
    (when (and buffer-file-name
               (string-match-p "/COMMIT_EDITMSG\\'" buffer-file-name))
      (erase-buffer)))

  (defun nl/git-commit-setup-cancel-safety ()
    "Empty COMMIT_EDITMSG before `with-editor-cancel' returns to Git."
    ;; `git-commit-save-message' is also on this hook; append this so the
    ;; message can still be saved to the message ring before the file is emptied.
    (add-hook 'with-editor-pre-cancel-hook
              #'nl/git-commit-empty-message-on-cancel t t))

  (defun nl/git-commit-setup-fonts ()
    "Make commit message buffers easier to read."
    (mixed-pitch-mode 1)
    (text-scale-set 1)
    (face-remap-add-relative 'git-commit-summary :inherit 'variable-pitch)
    (face-remap-add-relative 'git-commit-overlong-summary :inherit 'variable-pitch))

  (with-eval-after-load 'git-commit
    (add-hook 'git-commit-setup-hook #'nl/git-commit-setup-fill-paragraph)
    (add-hook 'git-commit-setup-hook #'nl/git-commit-setup-cancel-safety)
    (add-hook 'git-commit-setup-hook #'nl/git-commit-setup-fonts))
  (with-eval-after-load 'magit
    ;; A full Magit line background masks inline branch badge backgrounds, so
    ;; mark the current line with subtle rules instead.
    (set-face-attribute 'magit-section-highlight nil
                        :background 'unspecified
                        :foreground 'unspecified
                        :overline "#64748b"
                        :underline '(:color "#64748b" :style line)
                        :extend t)
    (let ((branch-font (face-attribute 'variable-pitch :family nil 'default)))
      (dolist (face '(magit-branch-current
                      magit-branch-local
                      magit-branch-remote
                      magit-branch-remote-head
                      magit-head))
        (let ((branch-background (pcase face
                                   ('magit-branch-current "#7c3aed")
                                   ('magit-branch-local "#2563eb")
                                   ('magit-branch-remote "#059669")
                                   ('magit-branch-remote-head "#0891b2")
                                   (_ "#6d28d9"))))
          (set-face-attribute face nil
                              :family branch-font
                              :foreground "#ffffff"
                              :background branch-background
                              :weight 'bold
                              :height 1.0
                              :box `(:line-width (1 . -1)
                                      :color ,branch-background
                                      :style released-button))))))
  (with-eval-after-load 'magit-status
    (define-key magit-status-mode-map (kbd "q") #'magit-quit-session)
    (magit-add-section-hook 'magit-status-sections-hook
                            'magit-insert-modules

                            'append))
  (setq-default vc-handled-backends '(Git))

**** Full-screen magit

The following code makes magit-status run alone in the frame, and then restores the old window configuration when you quit out of magit.

#+begin_src emacs-lisp
(define-advice magit-status (:around (orig-fun &rest args) magit-fullscreen)
  ;;(window-configuration-to-register ?magit-fullscreen)
  (window-configuration-to-register ?z)
  (apply orig-fun args)
  (delete-other-windows))

(defun magit-quit-session ()
  "Restores the previous window configuration and kills the magit buffer"
  (interactive)
  (kill-buffer)
  ;;(jump-to-register ?magit-fullscreen))
  (jump-to-register ?z))
#+end_src

*** [[https://github.com/10sr/git-walktree-el][git-walktree.el]]

Browse Git tree and blob objects

=M-x git-walktree= to open ~git-walktree~ buffer. Prompt user for revision to show.

When current buffer is visiting a file and it exists in the revision, open blob buffer for that file
Otherwise, open tree buffer for default-directory In ~git-walktree~ buffers, following keybinds are defined:

- ~^~ Go up to parent tree object
- ~P~ Move to parent revision
- ~N~ Move to child revision
- ~C~ Checkout current blob or tree object to working directory
- ~G~ Move to another revision (ask user)

Additionally, in tree object buffer:

- ~ENTER~ Visit tree or blob object at point

#+begin_src emacs-lisp :tangle no
(defun th/magit--with-difftastic (buffer command)
  "Run COMMAND with GIT_EXTERNAL_DIFF=difft then show result in BUFFER."
  (let ((process-environment
         (cons (concat "GIT_EXTERNAL_DIFF=difft --width="
                       (number-to-string (frame-width)))
               process-environment)))
    ;; Clear the result buffer (we might regenerate a diff, e.g., for
    ;; the current changes in our working directory).
    (with-current-buffer buffer
      (setq buffer-read-only nil)
      (erase-buffer))
    ;; Now spawn a process calling the git COMMAND.
    (make-process
     :name (buffer-name buffer)
     :buffer buffer
     :command command
     ;; Don't query for running processes when emacs is quit.
     :noquery t
     ;; Show the result buffer once the process has finished.
     :sentinel (lambda (proc event)
                 (when (eq (process-status proc) 'exit)
                   (with-current-buffer (process-buffer proc)
                     (goto-char (point-min))
                     (ansi-color-apply-on-region (point-min) (point-max))
                     (setq buffer-read-only t)
                     (view-mode)
                     (end-of-line)
                     ;; difftastic diffs are usually 2-column side-by-side,
                     ;; so ensure our window is wide enough.
                     (let ((width (current-column)))
                       (while (zerop (forward-line 1))
                         (end-of-line)
                         (setq width (max (current-column) width)))
                       ;; Add column size of fringes
                       (setq width (+ width
                                      (fringe-columns 'left)
                                      (fringe-columns 'right)))
                       (goto-char (point-min))
                       (pop-to-buffer
                        (current-buffer)
                        `(;; If the buffer is that wide that splitting the frame in
                          ;; two side-by-side windows would result in less than
                          ;; 80 columns left, ensure it's shown at the bottom.
                          ,(when (> 80 (- (frame-width) width))
                             #'display-buffer-at-bottom)
                          (window-width
                           . ,(min width (frame-width))))))))))))

(defun th/magit-show-with-difftastic (rev)
  "Show the result of \"git show REV\" with GIT_EXTERNAL_DIFF=difft."
  (interactive
   (list (or
          ;; If REV is given, just use it.
          (when (boundp 'rev) rev)
          ;; If not invoked with prefix arg, try to guess the REV from
          ;; point's position.
          (and (not current-prefix-arg)
               (or (magit-thing-at-point 'git-revision t)
                   (magit-branch-or-commit-at-point)))
          ;; Otherwise, query the user.
          (magit-read-branch-or-commit "Revision"))))
  (if (not rev)
      (error "No revision specified")
    (th/magit--with-difftastic
     (get-buffer-create (concat "*git show difftastic " rev "*"))
     (list "git" "--no-pager" "show" "--ext-diff" rev))))

(defun th/magit-diff-with-difftastic (arg)
  "Show the result of \"git diff ARG\" with GIT_EXTERNAL_DIFF=difft."
  (interactive
   (list (or
          ;; If RANGE is given, just use it.
          (when (boundp 'range) range)
          ;; If prefix arg is given, query the user.
          (and current-prefix-arg
               (magit-diff-read-range-or-commit "Range"))
          ;; Otherwise, auto-guess based on position of point, e.g., based on
          ;; if we are in the Staged or Unstaged section.
          (pcase (magit-diff--dwim)
            ('unmerged (error "unmerged is not yet implemented"))
            ('unstaged nil)
            ('staged "--cached")
            (`(stash . ,value) (error "stash is not yet implemented"))
            (`(commit . ,value) (format "%s^..%s" value value))
            ((and range (pred stringp)) range)
            (_ (magit-diff-read-range-or-commit "Range/Commit"))))))
  (let ((name (concat "*git diff difftastic"
                      (if arg (concat " " arg) "")
                      "*")))
    (th/magit--with-difftastic
     (get-buffer-create name)
     `("git" "--no-pager" "diff" "--ext-diff" ,@(when arg (list arg))))))

(transient-define-prefix th/magit-aux-commands ()

  ["Auxiliary commands"
   ("d" "Difftastic Diff (dwim)" th/magit-diff-with-difftastic)
   ("s" "Difftastic Show" th/magit-show-with-difftastic)])

(transient-append-suffix 'magit-dispatch "!"
