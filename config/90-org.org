(use-package org-modern
  :hook
  (org-mode . org-modern-mode)
  (org-agenda-finalize . org-modern-agenda)
  (org-modern-mode . variable-pitch-mode)
  :custom
  (org-modern-star '("◉" "○" "✸" "✿")) ;; Custom bullets for headlines
  (org-modern-list '((43 . "➤") (45 . "•") (42 . "◦"))) ;; Custom list markers
  (org-modern-table nil) ;; Disable table styling if not needed
  ;; NOTE: must not be `t'.  Hiding *all* stars marks the heading start
  ;; `invisible', which makes Org 9.8's `org-back-to-heading' raise
  ;; "Before first headline" and breaks TAB cycling.  `leading' keeps the
  ;; final star visible (still a clean look) and avoids the bug.
  (org-modern-hide-stars 'leading) ;; hide leading stars in headlines
  (org-modern-tag t) ;; Use org-modern labels for pill-like tag rendering
  (org-modern-label-border 4)
  (org-modern-block-fringe 2) ;; Add fringe to blocks
  (org-hide-emphasis-markers t) ;; hides ~ and =
  (org-modern-block-name '("«" . "»"))
  ;; org-modern renders keywords as pills from this list only.  A keyword missing
  ;; here keeps Org's ordinary TODO face even though TODO/DONE get the label font,
  ;; so every keyword in `org-todo-keywords' needs an entry.
  (org-modern-todo-faces
   '(("TODO"        . (:foreground "red" :weight bold))
     ("IN-PROGRESS" . (:foreground "orange" :weight bold))
     ("WAITING"     . (:foreground "magenta" :weight bold))
     ("DONE"        . (:foreground "green" :weight bold))
     ("CANCELLED"   . (:foreground "forest green" :weight bold))))
  (org-modern-priority
   '((?A . "⚡")
     (?B . "⬆")
     (?C . "⬇")))
  (org-modern-keyword
   '((t . (:foreground "gray" :weight bold))))
  (org-modern-horizontal-rule "────────────────────────────────────────")
  :config
  (set-face-attribute 'org-modern-tag nil
                      :background "#303030"
                      :foreground "#ffffff"
                      :inherit '(org-modern-label fixed-pitch))
  (set-face-attribute 'org-modern-label nil
                      :inherit '(fixed-pitch)
                      :height 0.9
                      :weight 'medium
                      :width 'normal)
  ;; Adjust line spacing only in org-mode buffers
  (defun nl/org-modern-line-spacing ()
    "Set line spacing specifically for org-mode buffers."
    (setq-local line-spacing 0.2)) ;; Adjust this value as needed

  (add-hook 'org-mode-hook #'nl/org-modern-line-spacing)
  )
