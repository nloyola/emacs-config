    (defun nl/apply-org-faces (&rest _)
      "Apply the preferred Org tag and monospaced-face styling.

The heights are relative (`nl/org-block-height'), never absolute: a
relative value multiplies the inherited `fixed-pitch', so it follows the
session's point size, while an absolute one would replace that height
outright and render at a fixed size regardless of the base font.

The one wrinkle is that Org buffers run `mixed-pitch-mode' (hooked on
text-mode in 20-fonts.org), which remaps `default' to `variable-pitch'.
Every relative height inside them therefore compounds by
`nl/variable-pitch-height' before it lands, so `fixed-pitch' - nominally
1.0 - really renders 1.28x the frame default.  Dividing that factor back
out here is what lets `nl/org-block-height' mean a plain multiple of the
frame `default', as its docstring claims.  Without it the old 0.85 was
really 0.85 * 1.28 = 1.09, and source blocks came out *larger* than the
default face instead of smaller.

These faces only ever render in Org buffers, where that remap is always
in effect, so the compensation is unconditional."
      (let ((block-height (/ nl/org-block-height nl/variable-pitch-height)))
        (set-face-attribute 'org-tag nil
                            :inherit '(fixed-pitch)
                            :background "#303030"
                            :foreground "#ffffff"
                            :box '(:line-width 2 :color "#505050"))
        (set-face-attribute 'org-table nil
                            :inherit 'fixed-pitch
                            :foreground "#91b831"

                            :inherit 'fixed-pitch
                            :background "gray18"

        ;; size follows the block they delimit.
        (dolist (face '(org-block-begin-line org-block-end-line
                        org-code org-verbatim))
          (when (facep face)
            (set-face-attribute face nil :height block-height)))))

*** HTML preview

One-shot preview of the current Org buffer in the browser, mirroring
=nl/markdown-preview= (=jc p= in Markdown buffers): export to a static HTML
file styled like GitHub's light theme, then open it once. Re-run to refresh.

#+begin_src emacs-lisp
  (defconst nl/org-github-css-file
    (expand-file-name "css/github-markdown-light.css" user-emacs-directory)
    "Vendored GitHub light-theme stylesheet used for the one-shot Org preview.")

  (defun nl/org-preview ()
    "Export the current Org buffer to a static HTML file and open it.
One-shot preview (no live reload): re-run to refresh.  The output file is
named after the source buffer so the browser URL ends in `<base>.html'
rather than a random temp name.  Each source file gets a stable path
(keyed by a hash of its name), so re-previewing overwrites in place
instead of littering the temp directory.

Uses `org-export-as' rather than `org-export-to-file' on purpose: the
latter is advised to run the email CSS inliner, which is irrelevant here."
    (interactive)
    (require 'ox-html)
    (let* ((source (or (buffer-file-name) (buffer-name)))
           (base (file-name-base source))
           (dir (expand-file-name
                 (md5 source)
                 (expand-file-name "emacs-org-preview" temporary-file-directory)))
           (html (expand-file-name (concat base ".html") dir))
           ;; The vendored stylesheet targets `.markdown-body', so put that
           ;; class on Org's content div and add GitHub's page-shell sizing
           ;; (which lives on github.com, not in the stylesheet itself).
           (org-html-content-class "markdown-body")
           (org-html-head-include-default-style nil)
           (org-html-head-include-scripts nil)
           (org-html-postamble nil)
           (org-html-head
            (concat "<link rel=\"stylesheet\" href=\"file://"
                    nl/org-github-css-file "\">"
                    "<style>"
                    "body { margin: 0; background: #ffffff; }"
                    ".markdown-body { box-sizing: border-box; min-width: 200px;"
                    " max-width: 980px; margin: 0 auto; padding: 45px; }"
                    "@media (max-width: 767px) { .markdown-body { padding: 15px; } }"
                    "</style>"))
           (output (org-export-as 'html)))
      (make-directory dir t)
      (with-temp-file html
        (insert output))
      (browse-url-of-file html)))

  (defun nl/org-command-menu ()
    "Open the Org command hydra."
    (interactive)
    (hydra-nl-org/body))

  (defun nl/org-mode-chords ()
    "Bind the Org command hydra to the `jc' key chord in this buffer."
    (key-chord-define-local "jc" #'nl/org-command-menu))

  (add-hook 'org-mode-hook #'nl/org-mode-chords)

  (with-eval-after-load 'hydra
    (defhydra hydra-nl-org (:color blue :hint nil)
      "Org"
      ("p" nl/org-preview "Render preview in browser" :column "Preview")
      ("e" org-html-export-to-html "Export to .html file" :column "Preview")))

  :hook

  :custom

  (org-modern-table nil) ;; Disable table styling if not needed
  ;; NOTE: must not be `t'.  Hiding *all* stars marks the heading start
  ;; `invisible', which makes Org 9.8's `org-back-to-heading' raise
  ;; "Before first headline" and breaks TAB cycling.  `leading' keeps the
  ;; final star visible (still a clean look) and avoids the bug.
  (org-modern-hide-stars 'leading) ;; hide leading stars in headlines
  (org-modern-tag t) ;; Use org-modern labels for pill-like tag rendering
  (org-modern-label-border 4)
  (org-modern-block-fringe 2) ;; Add fringe to blocks
  (org-hide-emphasis-markers t) ;; hides ~ and =
  (org-modern-block-name '("«" . "»"))
  ;; org-modern renders keywords as pills from this list only.  A keyword missing
  ;; here keeps Org's ordinary TODO face even though TODO/DONE get the label font,
  ;; so every keyword in `org-todo-keywords' needs an entry.
  (org-modern-todo-faces
   '(("TODO"        . (:foreground "red" :weight bold))
     ("IN-PROGRESS" . (:foreground "orange" :weight bold))
     ("WAITING"     . (:foreground "magenta" :weight bold))
     ("DONE"        . (:foreground "green" :weight bold))
     ("CANCELLED"   . (:foreground "forest green" :weight bold))))
  (org-modern-priority
   '((?A . "⚡")
     (?B . "⬆")
     (?C . "⬇")))
  (org-modern-keyword
   '((t . (:foreground "gray" :weight bold))))
  (org-modern-horizontal-rule "────────────────────────────────────────")
  :config
  (set-face-attribute 'org-modern-tag nil
                      :background "#303030"
                      :foreground "#ffffff"

*** org-roam-ui

Interactive D3 graph of the vault in the browser (=M-x org-roam-ui-open=,
served at http://localhost:35901). Replaces the deprecated org-roam-server.
=websocket= is the transport org-roam-ui talks to Emacs over.

#+begin_src emacs-lisp
(use-package websocket
  :after org-roam)

;; org-roam-ui pulls in simple-httpd. MELPA's simple-httpd recipe points at
;; `skeeto/emacs-web-server', whose basename collides with the unrelated
;; `eschulte/emacs-web-server' (the `web-server' package) already checked out
;; under elpaca/sources/emacs-web-server/. Elpaca then reuses that directory,
;; finds no simple-httpd.el, and the build fails ("Unable to find main elisp
;; file"). Pin simple-httpd to skeeto's original `emacs-http-server' repo,
;; which carries the same simple-httpd.el but has a non-colliding directory
;; name, so the dependency installs cleanly.
(use-package simple-httpd
  :ensure (simple-httpd :host github :repo "skeeto/emacs-http-server"))

(use-package org-roam-ui
