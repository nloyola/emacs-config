  (defun nl/php--comment-fill-prefix (line indentation)
    "Return fill prefix for LINE in a PHP comment with INDENTATION."
    (cond
     ((string-match-p (concat "^" (regexp-quote indentation) "\\* ?") line)
      (concat indentation "* "))
     ((string-match-p (concat "^" (regexp-quote indentation) "//+ ?") line)
      (concat indentation "// "))
     ((string-match-p (concat "^" (regexp-quote indentation) "# ?") line)
      (concat indentation "# "))
     (t indentation)))

  (defun nl/php--comment-line-text (line indentation)
    "Return comment body text from LINE with INDENTATION."
    (let ((trimmed (string-trim-left line)))
      (cond
       ((string-match "\\`/\\*+\\s-*\\(.*\\)\\'" trimmed)
        (match-string 1 trimmed))
       ((string-match "\\`\\*+/?\\s-*\\(.*\\)\\'" trimmed)
        (match-string 1 trimmed))
       ((string-match "\\`//+\\s-*\\(.*\\)\\'" trimmed)
        (match-string 1 trimmed))
       ((string-match "\\`#\\s-*\\(.*\\)\\'" trimmed)
        (match-string 1 trimmed))
       ((string-prefix-p indentation line)
        (string-remove-prefix indentation line))
       (t trimmed))))

  (defun nl/php-fill-paragraph (&optional justify)
    "Fill the current paragraph, including multiline PHP comments."
    (interactive "P")
    (if (not (nth 4 (syntax-ppss)))
        (let ((fill-paragraph-function nil))
          (fill-paragraph justify))
      (let* ((start (line-beginning-position))
             (end (line-end-position))
             (indentation (save-excursion
                            (back-to-indentation)
                            (buffer-substring-no-properties
                             (line-beginning-position)
                             (point))))
             (prefix (nl/php--comment-fill-prefix
                      (buffer-substring-no-properties start end)
                      indentation))
             (paragraph-start-line
              (save-excursion
                (while (and (> (line-beginning-position) (point-min))
                            (progn
                              (forward-line -1)
                              (nth 4 (syntax-ppss)))
                            (not (string-empty-p
                                  (string-trim
                                   (nl/php--comment-line-text
                                    (buffer-substring-no-properties
                                     (line-beginning-position)
                                     (line-end-position))
                                    indentation)))))
                  (setq start (line-beginning-position)))
                start))
             (paragraph-end-line
              (save-excursion
                (goto-char end)
                (while (and (= (forward-line 1) 0)
                            (nth 4 (syntax-ppss))
                            (not (string-empty-p
                                  (string-trim
                                   (nl/php--comment-line-text
                                    (buffer-substring-no-properties
                                     (line-beginning-position)
                                     (line-end-position))
                                    indentation)))))
                  (setq end (line-end-position)))
                end))
             (lines (split-string
                     (buffer-substring-no-properties paragraph-start-line paragraph-end-line)
                     "\n"))
             (text (string-join
                    (mapcar (lambda (line)
                              (string-trim
                               (nl/php--comment-line-text line indentation)))
                            lines)
                    " ")))
        (delete-region paragraph-start-line paragraph-end-line)
        (let ((fill-prefix prefix)
              (start-point (point)))
          (insert prefix text)
          (fill-region-as-paragraph start-point (point) justify))
        t)))

    "Return the heredoc/nowdoc node at point, or nil."
    (when (and (derived-mode-p 'php-ts-mode)
               (fboundp 'treesit-node-at))
      (let ((node (treesit-node-at (point) 'php))
            found)
        (while (and node (not found))
          (when (member (treesit-node-type node) '("heredoc" "nowdoc"))
            (setq found node))
          (setq node (treesit-node-parent node)))
        found)))

      (forward-line 0)
      (back-to-indentation)

    "Indent the current heredoc/NOWDOC line inside NODE."
    (let* ((base (nl/php--heredoc-base-indentation node))
           (offset php-ts-indent-offset)
           (target base))
      (save-excursion
        (forward-line -1)
        (while (and (> (point) (treesit-node-start node))
                    (looking-at-p "^[[:space:]]*$"))

    "Return a readable completion label for CANDIDATE."

  (defun nl/php--complete-lsp-symbol-at-point ()
    "Complete the current PHP symbol and apply LSP completion edits.
This handles the case where the symbol is already fully typed: Emacs'
built-in completion cycling can otherwise select the same text without
running the LSP completion exit function which inserts the matching `use'."
    (when (and (bound-and-true-p lsp-mode)
               (fboundp 'lsp-completion-at-point))
      (when-let* ((capf (lsp-completion-at-point))
                  (start (nth 0 capf))
                  (end (nth 1 capf))
                  (table (nth 2 capf))
                  (props (nthcdr 4 capf)))
        (let* ((prefix (buffer-substring-no-properties start end))
               (annotation-function (plist-get props :annotation-function))
               (exit-function (plist-get props :exit-function))
               (candidates (all-completions prefix table))
               (exact-candidates
                (seq-filter (lambda (candidate)
                              (string= prefix (substring-no-properties candidate)))
                            candidates)))
          (when (and exit-function exact-candidates)
            (let* ((choices
                    (mapcar (lambda (candidate)
                              (cons (nl/php--lsp-completion-label candidate annotation-function)
                                    candidate))
                            exact-candidates))
                   (choice
                    (if (= (length choices) 1)
                        (caar choices)
                      (completing-read "Import/complete PHP symbol: "
                                       (mapcar #'car choices) nil t nil nil
                                       (caar choices))))
                   (candidate (cdr (assoc choice choices))))
              (funcall exit-function candidate 'finished)
              t))))))

** rust

See https://robert.kra.hn/posts/rust-emacs-setup/

=rustic-mode= derives from =rust-mode=, which derives from =prog-mode= by
default and uses its own font-lock and indentation. Setting
=rust-mode-treesitter-derive= before =rust-mode= loads makes it derive from the
built-in =rust-ts-mode= instead, so =rustic-mode= inherits tree-sitter
font-lock and indentation while keeping cargo integration and the bindings
below. This needs the =rust= grammar from the Tree-sitter section in
=60-builtin.org=.
