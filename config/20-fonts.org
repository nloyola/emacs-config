(require 's)
(defvar nl/gui-fixed-font-name
  (when window-system
    (or
     (seq-first
      (seq-filter (lambda (font) (when (x-list-fonts font) font))
                  '("JetBrainsMono Nerd Font"
                    "CaskaydiaCove Nerd Font"
                    ;; funky font with litagures and a dotted 0
                    "Cascadia Code PL"
                    ;; clean font, but no litagures!?
                    "Hack Nerd Font"
                    "FiraCode Nerd Font"       ; has litagures
                    "Cousine Nerd Font"
                    "Iosevka Nerd Font"
                    "FantasqueSansMono Nerd Font"
                    "Monoid Nerd Font"
                    "Hasklig"
                    "Source Code Pro"
                    ;;
                    "IBM Plex Mono Medium"
                    "DejaVu Sans")))
     "monospaced"))
  "My fixed width font based on what I have installed.")

(defvar nl/gui-variable-font-name
  (when window-system
    (or
     (seq-first
      (seq-filter (lambda (font) (when (x-list-fonts font) font))
                  '("SN Pro"   ; https://supernotes.app/open-source/sn-pro

                    ;; Next best can be downloaded here:
                    ;; https://fontesk.com/xcharter-typeface/
                    "XCharter"
                    "Charter"
                    ;;  Interesting idea: "Iosevka Comfy Motion Duo"
                    "Serif"
                    ;;
                    "Alegreya")))
     (warn "Cannot find a Serif font. Install a serif font such as XCharter or Charter."))))

(defun nl/set-favorite-font-size (size)
  "Set the default font size as well as equalize the fixed and variable fonts."
  (let ((fav-font (format "%s-%d" nl/gui-fixed-font-name size)))
    (set-face-attribute 'default nil :font fav-font)
    (set-face-attribute 'fixed-pitch nil :family nl/gui-fixed-font-name :inherit 'default :height 120)
    (set-face-attribute 'variable-pitch nil :family nl/gui-variable-font-name :inherit 'default :height 170)))

(defun nl/mac-monitor-fontsize ()
  "Quickly set reset my font size when I connect my laptop to a monitor on a Mac."
  (interactive)
  (nl/set-favorite-font-size 13))

(defun nl/wayland-session-p ()
  "Return non-nil when this Emacs belongs to a Wayland session.

The X session runs xrandr --dpi 120 (see ~/.xprofile-asterix), so every
point size in this file renders at 1.25x under i3.  Wayland has no such
global knob and the heads are declared at scale 1, so the same point size
comes out ~20% smaller under Hyprland."
  (or (string= (getenv "XDG_SESSION_TYPE") "wayland")
      (and (getenv "WAYLAND_DISPLAY") t)))

(defun nl/linux-monitor-fontsize ()
  "Quickly set reset my font size when I connect my laptop to a monitor on Linux."
  (interactive)
  ;; 14 is 11 * 1.25 rounded: parity with what the X session renders.  Only the
  ;; default face follows this; the mode-line faces below stay at their fixed
  ;; :height 120.
  (nl/set-favorite-font-size (if (nl/wayland-session-p) 14 11)))

(defun nl/mac-laptop-fontsize ()
  "Quickly set reset my font size when I disconnect my laptop to a monitor from a Mac."
  (interactive)
  (nl/set-favorite-font-size 32))

(defun nl/linux-laptop-fontsize ()
  "Quickly set reset my font size when I disconnect my laptop to a monitor from Linux."
  (interactive)
  (nl/set-favorite-font-size 14))

(defun nl/imac-fontsize ()
  "Quickly set reset my font size when I am on my iMac."
  (interactive)
  (nl/set-favorite-font-size 16))

(defun nl/font-monitor-size-default ()
  "Set the default size according to my preference."
  (interactive)
  (cond
   ((s-starts-with? "gudrun" system-name) (nl/linux-laptop-fontsize))
   ((eq system-type 'gnu/linux)         (nl/linux-monitor-fontsize))
   (t                                   (nl/mac-monitor-fontsize))))

(when (display-graphic-p)
  (nl/font-monitor-size-default))

** [[https://gitlab.com/jabranham/mixed-pitch][Mixed Pitch]]

Mixed pitch is a minor mode that enables mixing fixed-pitch (also known as fixed-width or monospace)
and variable-pitch (AKA “proportional”) fonts. It tries to be smart about which fonts get which
face.

#+begin_src emacs-lisp
(use-package mixed-pitch
  :config
  (add-to-list 'mixed-pitch-fixed-pitch-faces 'org-property-value)
  (add-to-list 'mixed-pitch-fixed-pitch-faces 'font-lock-comment-face)
  :hook (text-mode . mixed-pitch-mode))
#+end_src

** Mode Line

Customize the mode line. This code has to run after init since setting a theme customize the mode line also.

See https://www.gonsie.com/blorg/modeline.html for additional details.

#+begin_src emacs-lisp
(use-package emacs
  :ensure nil
  :preface
  (defface nl/mode-line-fixed
    '((t (:height 120)))
    "Fixed-pitch text in the mode line.")

  (defface nl/mode-line-variable
    '((t (:height 120)))
    "Variable-pitch text in the mode line.")

  (defface nl/mode-line-buffer-id
    '((t (:foreground "goldenrod" :background "#565063" :height 120 :weight bold)))
    "Buffer name in the selected window's mode line.")

  (defface nl/mode-line-buffer-id-inactive
    '((t (:foreground "#9ca0b0" :background "#353644" :height 120)))
    "Buffer name in non-selected windows' mode lines.")

  (defface nl/mode-line-modified
    '((t (:slant italic :background "#565063" :height 120 :weight bold)))
    "Modified buffer name in the selected window's mode line.")

  (defface nl/mode-line-modified-inactive
    '((t (:foreground "#9ca0b0" :slant italic :background "#353644" :height 120)))
    "Modified buffer name in non-selected windows' mode lines.")

  (defun nl/mode-line-selected-window-p ()
    "Return non-nil when formatting the selected window's mode line."
    (if (fboundp 'mode-line-window-selected-p)
        (mode-line-window-selected-p)
      (eq (selected-window) (get-buffer-window (current-buffer)))))

  (defun nl/mode-line-buffer-face ()
    "Return the buffer-name face for the current mode line."
    (if (nl/mode-line-selected-window-p)
        (if (buffer-modified-p)
            'nl/mode-line-modified
          'nl/mode-line-buffer-id)
      (if (buffer-modified-p)
          'nl/mode-line-modified-inactive
        'nl/mode-line-buffer-id-inactive)))

  (defun nl/mode-line-theme ()
    (set-face-attribute 'mode-line nil
                        :background "#565063"
                        :foreground "white"
                        :box '(:line-width 2 :color "#3d3947")
                        :height 120
                        :overline nil
                        :underline nil
                        :family nl/gui-fixed-font-name
                        :inherit 'default)

    ;; Emacs 29+ uses `mode-line-active' for the selected window.  Some
    ;; themes make it inherit a large variable-pitch UI face, so set its
    ;; font attributes explicitly as well.
    (when (facep 'mode-line-active)
      (set-face-attribute 'mode-line-active nil
                          :background "#565063"
                          :foreground "white"
                          :box '(:line-width 2 :color "goldenrod")
                          :height 120
                          :overline nil
                          :underline nil
                          :family nl/gui-fixed-font-name
                          :inherit 'default))

    (set-face-attribute 'mode-line-inactive nil
                        :background "#353644"
                        :foreground "#9ca0b0"
                        :box '(:line-width 2 :color "#242631")
                        :height 120
                        :overline nil
                        :family nl/gui-fixed-font-name
                        :inherit 'default)

    (set-face-attribute 'nl/mode-line-fixed nil
                        :height 120
                        :family nl/gui-fixed-font-name
                        :inherit nil)
    (set-face-attribute 'nl/mode-line-variable nil
                        :height 120
                        :family nl/gui-variable-font-name
                        :inherit nil)
    (set-face-attribute 'nl/mode-line-buffer-id nil
                        :foreground "goldenrod"
                        :background "#565063"
                        :weight 'bold
                        :height 120
                        :family nl/gui-fixed-font-name
                        :inherit nil)
    (set-face-attribute 'nl/mode-line-buffer-id-inactive nil
                        :foreground "#9ca0b0"
                        :background "#353644"
                        :weight 'normal
                        :height 120
                        :family nl/gui-fixed-font-name
                        :inherit nil)
    (set-face-attribute 'nl/mode-line-modified nil
                        :slant 'italic
                        :background "#565063"
                        :weight 'bold
                        :height 120
                        :family nl/gui-fixed-font-name
                        :inherit nil)
    (set-face-attribute 'nl/mode-line-modified-inactive nil
                        :foreground "#9ca0b0"
                        :slant 'italic
                        :background "#353644"
                        :weight 'normal
                        :height 120
                        :family nl/gui-fixed-font-name
                        :inherit nil))
  :init
  (setq-default
   mode-line-format
   (list
    ;; the buffer name; the file name as a tool tip
    '(:eval (propertize (if (nl/mode-line-selected-window-p)
                            " ◆ %b "
                          "   %b ")
                        'face (nl/mode-line-buffer-face)
                        'help-echo (buffer-file-name)))

    ;; line and column
    " ["
    (propertize "%l" 'face '(:inherit nl/mode-line-fixed :foreground "light sky blue")) ","
    (propertize "%c" 'face '(:inherit nl/mode-line-fixed :foreground "light green"))
    "] "

    ;; relative position, size of file
    " ("
    (propertize "%p" 'face '(:inherit nl/mode-line-fixed :foreground "salmon")) ;; % above top
    "/"
    (propertize "%I" 'face '(:inherit nl/mode-line-fixed :foreground "sandy brown")) ;; size
    ") "

    ;; ansi-term's line or char mode
    ":"
    'mode-line-process
    ":"

    ;; spaces to align right
    '(:eval (propertize
             " " 'display
             `((space :align-to (- (+ right right-fringe right-margin)
                                   ,(+ 50 (string-width (if (listp mode-name) (car mode-name) mode-name))))))))

    ;; the current major mode
    (propertize " %m " 'face '(:inherit nl/mode-line-variable :foreground "light sky blue"))
    minor-mode-alist
    ))
  :hook (after-init . nl/mode-line-theme))
#+end_src
