(defvar nl/fixed-pitch-height 1.0
  "Relative `:height' for the `fixed-pitch' face, as a multiple of `default'.

Org source blocks, tables, inline code, everything `mixed-pitch' keeps
monospaced, and the line numbers render at this size - Modus with
`modus-themes-mixed-fonts' gives `line-number' :inherit (fixed-pitch
default), so this is the one knob for all of them.

Deliberately relative, not absolute: a relative height multiplies the
inherited `default', which already carries the session's point size, so
it needs no per-session DPI compensation.  An absolute value here
renders larger than the code it sits beside under Wayland - that is what
put the line numbers at 21px against 18px code.")

(defvar nl/variable-pitch-height 1.28
  "Relative `:height' for the `variable-pitch' face, as a multiple of `default'.

Org body text renders at this size, and the theme's heading scales
\(see `modus-themes-headings' in 50-theme.org) multiply it, so lowering
this shrinks the headings along with the prose.

Relative for the same reason as `nl/fixed-pitch-height': it multiplies
the inherited `default', so it tracks the session's point size instead
of needing a Wayland fudge factor.")

(defvar nl/org-block-height 1.0
  "Relative `:height' for Org's monospaced faces, as a multiple of `default'.

Applied by `nl/apply-org-faces' in 90-org.org to `org-block', its
begin/end lines, `org-table', `org-code' and `org-verbatim'.

1.0 means an Org source block renders at exactly the size of code in a
prog-mode buffer - and, since `nl/linux-monitor-fontsize' is kept pinned
to Alacritty's cell, at the size Alacritty renders too.

Read this as a multiple of the frame `default', NOT of the height Emacs
finally applies.  Org buffers run `mixed-pitch-mode', which remaps
`default' to `variable-pitch', so every relative height in them is
multiplied by `nl/variable-pitch-height' before it lands.
`nl/apply-org-faces' divides that factor back out so the number here
means what it says.  That compensation is why this is 1.0 and not the
0.85 it used to be: 0.85 was really 0.85 * 1.28 = 1.09, which rendered
source blocks larger than the default face rather than smaller.

Separate from `nl/fixed-pitch-height' on purpose: `line-number' inherits
`fixed-pitch', so shrinking that face to taste inside Org would drag the
line numbers out of step with the code they number.")

(defun nl/set-favorite-font (font)
  "Set FONT as the `default' face and equalize the fixed and variable fonts.

FONT is anything Emacs can parse as a font name.  The two callers below
build one from a point size or from a pixel size; nothing else should
call this directly."
  (set-face-attribute 'default nil :font font)
  ;; Relative height - no `nl/font-height' here, see `nl/fixed-pitch-height'.
  (set-face-attribute 'fixed-pitch nil :family nl/gui-fixed-font-name :inherit 'default
                      :height nl/fixed-pitch-height)
  ;; Relative height - see `nl/variable-pitch-height'.
  (set-face-attribute 'variable-pitch nil :family nl/gui-variable-font-name :inherit 'default
                      :height nl/variable-pitch-height))

(defun nl/set-favorite-font-size (size)
  "Set the default font to SIZE points, equalizing the pitch faces.

A point size renders through whatever DPI the build resolved, so it is
only as stable as that DPI is.  Under Wayland prefer
`nl/set-favorite-font-pixel-size'.

%s, not %d: a size may be fractional and %d would silently truncate it.
Integer sizes still format identically."
  (nl/set-favorite-font (format "%s-%s" nl/gui-fixed-font-name size)))

(defun nl/set-favorite-font-pixel-size (pixels)
  "Set the default font to PIXELS pixels tall, equalizing the pitch faces.

Pixels, not points, because this pgtk build turns a point size into
pixels with `gdk_screen_get_resolution' - sampled once when the first
graphical frame is created, never updated, and fed on Wayland by
`org.gnome.desktop.interface text-scaling-factor' * 96.  A point size
therefore made Emacs a consumer of a desktop-wide key: raising it for
this frame shrank every point size the rest of the session was written
in, and Chrome's tab strip in particular, since Chrome reads the GTK
font in points but does not apply that key to it.

Pinning the size in pixels takes the key out of the loop, so it can hold
whatever value suits the rest of the desktop and Emacs still renders the
same on any DPI - including the 96 fallback a frame gets when the
settings portal is not answering yet at session start."
  (nl/set-favorite-font (format "%s:pixelsize=%d" nl/gui-fixed-font-name pixels)))

(defun nl/mac-monitor-fontsize ()
  "Quickly set reset my font size when I connect my laptop to a monitor on a Mac."
  (interactive)
  (nl/set-favorite-font-size 11))

(defun nl/wayland-session-p ()
  "Return non-nil when this Emacs belongs to a Wayland session.

The X session runs xrandr --dpi 120 (see ~/.xprofile-asterix), so every
point size in this file renders at 1.25x under i3.  Wayland has no such
global knob and the heads are declared at scale 1, so the same point size
comes out ~20% smaller under Hyprland."
  (or (string= (getenv "XDG_SESSION_TYPE") "wayland")
      (and (getenv "WAYLAND_DISPLAY") t)))

(defun nl/linux-monitor-fontsize ()
  "Quickly set reset my font size when I connect my laptop to a monitor on Linux."
  (interactive)
  ;; The Wayland size matches Alacritty, which is the reference now that
  ;; alacritty.toml carries its own Hyprland size (13).  Measured, not
  ;; derived: Alacritty reports 189x43 cells in an 1894x1034 window with 6px
  ;; padding, so its cell is 9.93x23.5px, and the target is the 17px font /
  ;; 10x24 cell that sits on top of it.
  ;;
  ;; Stated in pixels, which is the only unit both ends of that comparison
  ;; agree on: Alacritty reads the Wayland output scale (1.0 here, so 96 DPI)
  ;; and ignores `text-scaling-factor', while Emacs resolves a point size
  ;; through that key - see `nl/set-favorite-font-pixel-size'.  Equal point
  ;; sizes therefore did not render equal, and the fix used to be a point
  ;; size chosen per DPI (10.5 at 1.2, 7.6 at 1.6875), which had to be
  ;; re-derived every time the key moved.  17px is 17px at any DPI.
  ;;
  ;; Re-measure with `font-info' and `frame-char-width' if the target
  ;; changes; want 17, 10, 24.
  ;;
  ;; Replaces 13 (= 11 * 1.25), which was chosen to match the X session's own
  ;; Emacs rather than the terminal, and rendered ~31% wider than Alacritty.
  ;; Only the default face follows this; the mode-line faces below are sized
  ;; from it, see `nl/mode-line-height'.
  (if (nl/wayland-session-p)
      (nl/set-favorite-font-pixel-size 17)
    (nl/set-favorite-font-size 11)))

** [[https://gitlab.com/jabranham/mixed-pitch][Mixed Pitch]]

Mixed pitch is a minor mode that enables mixing fixed-pitch (also known as fixed-width or monospace)
and variable-pitch (AKA “proportional”) fonts. It tries to be smart about which fonts get which
face.

#+begin_src emacs-lisp
(use-package mixed-pitch
  :config
  (add-to-list 'mixed-pitch-fixed-pitch-faces 'org-property-value)
  (add-to-list 'mixed-pitch-fixed-pitch-faces 'font-lock-comment-face)
  :hook (text-mode . mixed-pitch-mode))
#+end_src

** Mode Line

Customize the mode line. This code has to run after init since setting a theme customize the mode line also.

See https://www.gonsie.com/blorg/modeline.html for additional details.

#+begin_src emacs-lisp
(use-package emacs
  :ensure nil
  :preface
  (defconst nl/mode-line-pixel-height 19
    "Height in pixels of every mode-line face.

Sized to sit ~12% above the 17px `default' face, which is the
relationship this mode line has always had.")

  (defun nl/mode-line-height ()
    "Return the absolute `:height' (1/10 pt) for every mode-line face.

Absolute, not relative, because these faces do not share a base: some
set `:inherit nil' and others inherit `mode-line-buffer-id', so a float
would multiply something different in each and they would not come out
the same size.

An absolute height is in 1/10 pt, so it resolves through the DPI the
build sampled at first-frame creation - which is why a literal value
here was wrong at every DPI but the one it was written for (120 renders
27px at 162 DPI, towering over a 17px default).  The `default' face is
pinned in pixels instead (see `nl/set-favorite-font-pixel-size'), so its
own `:height' is this session's measured px-to-point conversion, and
scaling that by the pixel ratio lands on the right point size at any
DPI.

Falls back to 120 - correct at 115.2 DPI - when there is no font to
measure, which is the case in a frameless daemon."
    (let* ((info (ignore-errors (font-info (face-font 'default))))
           (px (and info (aref info 2))))
      (if (and px (> px 0))
          (max 1 (round (* (face-attribute 'default :height)
                           (/ (float nl/mode-line-pixel-height) px))))
        120)))

  ;; None of these faces names a colour, on purpose.  Emacs draws the mode
  ;; line with `mode-line' (selected window) or `mode-line-inactive' (any
  ;; other) as the base face and merges the propertized faces on top, so a
  ;; face that leaves foreground and background unspecified inherits whatever
  ;; the active theme put there - and the active/inactive distinction comes
  ;; out for free, without a second set of faces to keep in step.
  ;;
  ;; That matters here because `qs-theme' switches Emacs across 25 palettes
  ;; (see the omarchy block in 50-theme.org).  Any literal colour would be
  ;; wrong in 24 of them.  It also defeats the point of the tritanopia
  ;; palettes, which avoid exactly the blue/yellow distinctions the old
  ;; goldenrod-on-#565063 mode line was built out of.

  (defface nl/mode-line-fixed
    `((t (:height ,(nl/mode-line-height) :inherit nil)))
    "Fixed-pitch text in the mode line.
Colourless by design; see the commentary above.")

  (defface nl/mode-line-variable
    `((t (:height ,(nl/mode-line-height) :inherit nil)))
    "Variable-pitch text in the mode line.
Colourless by design; see the commentary above.")

  (defface nl/mode-line-buffer-id
    `((t (:height ,(nl/mode-line-height) :weight bold :inherit mode-line-buffer-id)))
    "Buffer name in the selected window's mode line.
Inherits `mode-line-buffer-id', which every theme styles.")

  (defface nl/mode-line-buffer-id-inactive
    `((t (:height ,(nl/mode-line-height) :weight normal :inherit nil)))
    "Buffer name in non-selected windows' mode lines.
Nothing to dim by hand: the base `mode-line-inactive' already is.")

  (defface nl/mode-line-modified
    `((t (:height ,(nl/mode-line-height) :weight bold :slant italic
          :inherit mode-line-buffer-id)))
    "Modified buffer name in the selected window's mode line.
Italic is the modified marker - there is no standard face for this state,
and a borrowed one (`warning', say) is toned for `bg-main' rather than for
a mode-line background.")

  (defface nl/mode-line-modified-inactive
    `((t (:height ,(nl/mode-line-height) :weight normal :slant italic :inherit nil)))
    "Modified buffer name in non-selected windows' mode lines.")

  (defun nl/mode-line-selected-window-p ()
    "Return non-nil when formatting the selected window's mode line."
    (if (fboundp 'mode-line-window-selected-p)
        (mode-line-window-selected-p)
      (eq (selected-window) (get-buffer-window (current-buffer)))))

  (defun nl/mode-line-buffer-face ()
    "Return the buffer-name face for the current mode line."
    (if (nl/mode-line-selected-window-p)
        (if (buffer-modified-p)
            'nl/mode-line-modified
          'nl/mode-line-buffer-id)
      (if (buffer-modified-p)
          'nl/mode-line-modified-inactive
        'nl/mode-line-buffer-id-inactive)))

  (defun nl/mode-line-theme (&optional _theme)
    "Re-apply the mode line's font size on top of the active theme.

Size only - the colours, box and family of `mode-line', `mode-line-active'
and `mode-line-inactive' are left to whichever theme is loaded.  Without
this the mode line would render at the theme's own UI size (22px under
`modus-themes-variable-pitch-ui'), which towers over a 17px default and
over the 19px `nl/mode-line-*' segments drawn on top of it.

Takes and ignores an argument so it can go straight on
`enable-theme-functions', which calls its hooks with the theme."
    (dolist (face '(mode-line mode-line-active mode-line-inactive))
      (when (facep face)
        (set-face-attribute face nil :height (nl/mode-line-height))))

    ;; The `nl/mode-line-*' faces are ours, so no theme overwrites them and
    ;; their defface specs stand.  Re-asserting the two attributes that
    ;; depend on variables keeps a `C-M-x' on this file enough to see a font
    ;; change, rather than needing a restart.
    (dolist (face '(nl/mode-line-fixed nl/mode-line-buffer-id
                    nl/mode-line-buffer-id-inactive nl/mode-line-modified
                    nl/mode-line-modified-inactive))
      (set-face-attribute face nil
                          :height (nl/mode-line-height)
                          :family nl/gui-fixed-font-name))
    (set-face-attribute 'nl/mode-line-variable nil
                        :height (nl/mode-line-height)
                        :family nl/gui-variable-font-name))
  :init
  (setq-default
   mode-line-format
   (list
    ;; the buffer name; the file name as a tool tip
    '(:eval (propertize (if (nl/mode-line-selected-window-p)
                            " ◆ %b "
                          "   %b ")
                        'face (nl/mode-line-buffer-face)
                        'help-echo (buffer-file-name)))

    ;; line and column
    " ["
    (propertize "%l" 'face 'nl/mode-line-fixed) ","
    (propertize "%c" 'face 'nl/mode-line-fixed)
    "] "

    ;; relative position, size of file
    " ("
    ;; Undimmed on purpose.  `shadow' is the obvious face for a secondary
    ;; readout, but Modus tunes it for `bg-main', not for the mode line:
    ;; against `bg-mode-line-active' (#003c52 here) its #989898 measures
    ;; 4.12:1, under the 4.5:1 WCAG AA needs for text this size.  The plain
    ;; mode-line foreground is 11.88:1, and every theme guarantees its own
    ;; mode-line contrast, so this stays legible across all 25 palettes.
    (propertize "%p" 'face 'nl/mode-line-fixed) ;; % above top
    "/"
    (propertize "%I" 'face 'nl/mode-line-fixed) ;; size
    ") "

    ;; ansi-term's line or char mode
    ":"
    'mode-line-process
    ":"

    ;; spaces to align right
    '(:eval (propertize
             " " 'display
             `((space :align-to (- (+ right right-fringe right-margin)
                                   ,(+ 50 (string-width (if (listp mode-name) (car mode-name) mode-name))))))))

    ;; the current major mode
    (propertize " %m " 'face '(:inherit (mode-line-emphasis nl/mode-line-variable)))
    minor-mode-alist
    ))
  :config
  ;; NOT `after-init'.  omarchy is an elpaca package, and elpaca activates
  ;; packages asynchronously, so the `load-theme' inside `nl/qs-theme-sync'
  ;; (50-theme.org) runs *after* `after-init-hook' has already fired - the
  ;; theme won every startup and this function's work was invisible.  The
  ;; same applies at runtime: `qs-theme set' calls back over emacsclient and
  ;; loads a theme whenever the desktop palette changes.
  ;;
  ;; `enable-theme-functions' fires on every one of those, which is exactly
  ;; when the mode-line faces need re-asserting.
  (add-hook 'enable-theme-functions #'nl/mode-line-theme)
  ;; ...and once now, for the case where a theme is already in force.
  (nl/mode-line-theme))
#+end_src
