(defvar nl/fixed-pitch-height 1.0
  "Relative `:height' for the `fixed-pitch' face, as a multiple of `default'.

Org source blocks, tables, inline code, everything `mixed-pitch' keeps
monospaced, and the line numbers render at this size - Modus with
`modus-themes-mixed-fonts' gives `line-number' :inherit (fixed-pitch
default), so this is the one knob for all of them.

Deliberately relative, not absolute: a relative height multiplies the
inherited `default', which already carries the session's point size, so
it needs no per-session DPI compensation.  An absolute value here
renders larger than the code it sits beside under Wayland - that is what
put the line numbers at 21px against 18px code.")

(defvar nl/variable-pitch-height 1.28
  "Relative `:height' for the `variable-pitch' face, as a multiple of `default'.

Org body text renders at this size, and the theme's heading scales
\(see `modus-themes-headings' in 50-theme.org) multiply it, so lowering
this shrinks the headings along with the prose.

Relative for the same reason as `nl/fixed-pitch-height': it multiplies
the inherited `default', so it tracks the session's point size instead
of needing a Wayland fudge factor.")

(defvar nl/org-block-height 1.0
  "Relative `:height' for Org's monospaced faces, as a multiple of `default'.

Applied by `nl/apply-org-faces' in 90-org.org to `org-block', its
begin/end lines, `org-table', `org-code' and `org-verbatim'.

1.0 means an Org source block renders at exactly the size of code in a
prog-mode buffer - and, since `nl/linux-monitor-fontsize' is kept pinned
to Alacritty's cell, at the size Alacritty renders too.

Read this as a multiple of the frame `default', NOT of the height Emacs
finally applies.  Org buffers run `mixed-pitch-mode', which remaps
`default' to `variable-pitch', so every relative height in them is
multiplied by `nl/variable-pitch-height' before it lands.
`nl/apply-org-faces' divides that factor back out so the number here
means what it says.  That compensation is why this is 1.0 and not the
0.85 it used to be: 0.85 was really 0.85 * 1.28 = 1.09, which rendered
source blocks larger than the default face rather than smaller.

Separate from `nl/fixed-pitch-height' on purpose: `line-number' inherits
`fixed-pitch', so shrinking that face to taste inside Org would drag the
line numbers out of step with the code they number.")

(defun nl/set-favorite-font-size (size)
  "Set the default font size as well as equalize the fixed and variable fonts."
  ;; %s, not %d: the Wayland size is fractional (7.6) and %d would silently
  ;; truncate it to 7.  Integer sizes still format identically.
  (let ((fav-font (format "%s-%s" nl/gui-fixed-font-name size)))
    (set-face-attribute 'default nil :font fav-font)
    ;; Relative height - no `nl/font-height' here, see `nl/fixed-pitch-height'.
    (set-face-attribute 'fixed-pitch nil :family nl/gui-fixed-font-name :inherit 'default
                        :height nl/fixed-pitch-height)
    ;; Relative height - see `nl/variable-pitch-height'.
    (set-face-attribute 'variable-pitch nil :family nl/gui-variable-font-name :inherit 'default
                        :height nl/variable-pitch-height)))

(defun nl/mac-monitor-fontsize ()
  "Quickly set reset my font size when I connect my laptop to a monitor on a Mac."
  (interactive)
  (nl/set-favorite-font-size 11))

(defun nl/wayland-session-p ()
  "Return non-nil when this Emacs belongs to a Wayland session.

The X session runs xrandr --dpi 120 (see ~/.xprofile-asterix), so every
point size in this file renders at 1.25x under i3.  Wayland has no such
global knob and the heads are declared at scale 1, so the same point size
comes out ~20% smaller under Hyprland."
  (or (string= (getenv "XDG_SESSION_TYPE") "wayland")
      (and (getenv "WAYLAND_DISPLAY") t)))

(defun nl/linux-monitor-fontsize ()
  "Quickly set reset my font size when I connect my laptop to a monitor on Linux."
  (interactive)
  ;; The Wayland size matches Alacritty, which is the reference now that
  ;; alacritty.toml carries its own Hyprland size (13).  Measured, not
  ;; derived: Alacritty reports 189x43 cells in an 1894x1034 window with 6px
  ;; padding, so its cell is 9.93x23.5px, and the target is the 17px font /
  ;; 10x24 cell that sits on top of it.
  ;;
  ;; The number is a function of the DPI this build resolves, which is
  ;; `org.gnome.desktop.interface text-scaling-factor' times 96, sampled once
  ;; when the first graphical frame is created and never updated (this pgtk
  ;; build is configured --without-gsettings, so nothing tracks the key).
  ;; Alacritty ignores that key entirely - it reads the Wayland output scale,
  ;; which is 1.0 - so it stays at 96 DPI no matter what this is set to, and
  ;; equal point sizes do not render equal.  Cell size is the only reliable
  ;; way to line the two up.
  ;;
  ;;   text-scaling-factor   DPI     size for a 10x24 cell
  ;;   1.2                   115.2   10.5
  ;;   1.6875 (162, actual)  162     7.6   <- current
  ;;
  ;; 7.6 is the middle of the 7.4-7.8 plateau that all render 17px, so DPI
  ;; drift does not tip it to 16 or 18.  Re-measure with `font-info' and
  ;; `frame-char-width' if text-scaling-factor changes again.
  ;;
  ;; Replaces 13 (= 11 * 1.25), which was chosen to match the X session's own
  ;; Emacs rather than the terminal, and rendered ~31% wider than Alacritty.
  ;; Only the default face follows this; the mode-line faces below carry
  ;; their own absolute size, `nl/mode-line-height'.
  (nl/set-favorite-font-size (if (nl/wayland-session-p) 7.6 11)))

  "Set the default size according to my preference."

** [[https://gitlab.com/jabranham/mixed-pitch][Mixed Pitch]]

Mixed pitch is a minor mode that enables mixing fixed-pitch (also known as fixed-width or monospace)
and variable-pitch (AKA “proportional”) fonts. It tries to be smart about which fonts get which
face.

#+begin_src emacs-lisp
(use-package mixed-pitch
  :config
  (add-to-list 'mixed-pitch-fixed-pitch-faces 'org-property-value)
  (add-to-list 'mixed-pitch-fixed-pitch-faces 'font-lock-comment-face)
  :hook (text-mode . mixed-pitch-mode))
#+end_src

** Mode Line

Customize the mode line. This code has to run after init since setting a theme customize the mode line also.

See https://www.gonsie.com/blorg/modeline.html for additional details.

#+begin_src emacs-lisp
(use-package emacs
  :ensure nil
  :preface
  (defconst nl/mode-line-height 85
    "Absolute `:height' (1/10 pt) for every mode-line face.

Absolute, not relative, because the `nl/mode-line-*' faces set
`:inherit nil' so a float here would have no `default' to multiply.
That makes it DPI-dependent: Emacs resolves it through the DPI sampled
at first-frame creation, which is `text-scaling-factor' * 96 (see
`nl/linux-monitor-fontsize').

Sized to sit ~12% above the `default' face, which is the relationship
this mode line has always had:

  text-scaling-factor   DPI     default   this   renders
  1.0                   96      14px      120    16px
  1.2                   115.2   17px      120    19px
  1.6875                162     17px       85    19px   <- current

Re-measure with `font-info' on `mode-line' if the DPI changes again;
120 resolves to 27px at 162 DPI, which towers over a 17px default.")

  (defface nl/mode-line-fixed
    `((t (:height ,nl/mode-line-height)))
    "Fixed-pitch text in the mode line.")

  (defface nl/mode-line-variable
    `((t (:height ,nl/mode-line-height)))
    "Variable-pitch text in the mode line.")

  (defface nl/mode-line-buffer-id
    `((t (:foreground "goldenrod" :background "#565063" :height ,nl/mode-line-height :weight bold)))
    "Buffer name in the selected window's mode line.")

  (defface nl/mode-line-buffer-id-inactive
    `((t (:foreground "#9ca0b0" :background "#353644" :height ,nl/mode-line-height)))
    "Buffer name in non-selected windows' mode lines.")

  (defface nl/mode-line-modified
    `((t (:slant italic :background "#565063" :height ,nl/mode-line-height :weight bold)))
    "Modified buffer name in the selected window's mode line.")

  (defface nl/mode-line-modified-inactive
    `((t (:foreground "#9ca0b0" :slant italic :background "#353644" :height ,nl/mode-line-height)))
    "Modified buffer name in non-selected windows' mode lines.")

  (defun nl/mode-line-selected-window-p ()
    "Return non-nil when formatting the selected window's mode line."
    (if (fboundp 'mode-line-window-selected-p)
        (mode-line-window-selected-p)
      (eq (selected-window) (get-buffer-window (current-buffer)))))

  (defun nl/mode-line-buffer-face ()
    "Return the buffer-name face for the current mode line."
    (if (nl/mode-line-selected-window-p)
        (if (buffer-modified-p)
            'nl/mode-line-modified
          'nl/mode-line-buffer-id)
      (if (buffer-modified-p)
          'nl/mode-line-modified-inactive
        'nl/mode-line-buffer-id-inactive)))

  (defun nl/mode-line-theme ()
    (set-face-attribute 'mode-line nil
                        :background "#565063"
                        :foreground "white"
                        :box '(:line-width 2 :color "#3d3947")
                        :height nl/mode-line-height
                        :overline nil
                        :underline nil
                        :family nl/gui-fixed-font-name
                        :inherit 'default)

    ;; Emacs 29+ uses `mode-line-active' for the selected window.  Some
    ;; themes make it inherit a large variable-pitch UI face, so set its
    ;; font attributes explicitly as well.
    (when (facep 'mode-line-active)
      (set-face-attribute 'mode-line-active nil
                          :background "#565063"
                          :foreground "white"
                          :box '(:line-width 2 :color "goldenrod")
                          :height nl/mode-line-height
                          :overline nil
                          :underline nil
                          :family nl/gui-fixed-font-name
                          :inherit 'default))

    (set-face-attribute 'mode-line-inactive nil
                        :background "#353644"
                        :foreground "#9ca0b0"
                        :box '(:line-width 2 :color "#242631")
                        :height nl/mode-line-height
                        :overline nil
                        :family nl/gui-fixed-font-name
                        :inherit 'default)

    (set-face-attribute 'nl/mode-line-fixed nil
                        :height nl/mode-line-height
                        :family nl/gui-fixed-font-name
                        :inherit nil)
    (set-face-attribute 'nl/mode-line-variable nil
                        :height nl/mode-line-height
                        :family nl/gui-variable-font-name
                        :inherit nil)
    (set-face-attribute 'nl/mode-line-buffer-id nil
                        :foreground "goldenrod"
                        :background "#565063"
                        :weight 'bold
                        :height nl/mode-line-height
                        :family nl/gui-fixed-font-name
                        :inherit nil)
    (set-face-attribute 'nl/mode-line-buffer-id-inactive nil
                        :foreground "#9ca0b0"
                        :background "#353644"
                        :weight 'normal
                        :height nl/mode-line-height
                        :family nl/gui-fixed-font-name
                        :inherit nil)
    (set-face-attribute 'nl/mode-line-modified nil
                        :slant 'italic
                        :background "#565063"
                        :weight 'bold
                        :height nl/mode-line-height
                        :family nl/gui-fixed-font-name
                        :inherit nil)
    (set-face-attribute 'nl/mode-line-modified-inactive nil
                        :foreground "#9ca0b0"
                        :slant 'italic
                        :background "#353644"
                        :weight 'normal
                        :height nl/mode-line-height
                        :family nl/gui-fixed-font-name
                        :inherit nil))
  :init
  (setq-default
   mode-line-format
   (list
    ;; the buffer name; the file name as a tool tip
    '(:eval (propertize (if (nl/mode-line-selected-window-p)
                            " ◆ %b "
                          "   %b ")
                        'face (nl/mode-line-buffer-face)
                        'help-echo (buffer-file-name)))

    ;; line and column
    " ["
    (propertize "%l" 'face '(:inherit nl/mode-line-fixed :foreground "light sky blue")) ","
    (propertize "%c" 'face '(:inherit nl/mode-line-fixed :foreground "light green"))
    "] "

    ;; relative position, size of file
    " ("
    (propertize "%p" 'face '(:inherit nl/mode-line-fixed :foreground "salmon")) ;; % above top
    "/"
    (propertize "%I" 'face '(:inherit nl/mode-line-fixed :foreground "sandy brown")) ;; size
    ") "

    ;; ansi-term's line or char mode
    ":"
    'mode-line-process
    ":"

    ;; spaces to align right
    '(:eval (propertize
             " " 'display
             `((space :align-to (- (+ right right-fringe right-margin)
                                   ,(+ 50 (string-width (if (listp mode-name) (car mode-name) mode-name))))))))

    ;; the current major mode
    (propertize " %m " 'face '(:inherit nl/mode-line-variable :foreground "light sky blue"))
    minor-mode-alist
    ))
  :hook (after-init . nl/mode-line-theme))
#+end_src
