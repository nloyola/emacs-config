(use-package keycast
  :ensure t
  :config
  ;; This config replaces `mode-line-buffer-identification`, so keycast
  ;; needs to insert after a segment that actually exists.
  (setq keycast-mode-line-insert-after 'mode-line-process)

  ;; This enables the display in the mode-line (bottom bar)
  (keycast-mode-line-mode 1)

  ;; Optional: If you want a more prominent display,
  ;; you can use the header-line (top of the window) instead:
  ;; (keycast-header-line-mode 1)

  ;; Optional: Control what exactly is shown (keys + command name)
  (setq keycast-substitute-alist '((self-insert-command nil nil))))
