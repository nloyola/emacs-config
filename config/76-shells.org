(use-package eat
  ;; ensure the eat entry points are autoloaded
  :commands (eat eat-other-window)
  :custom-face
  (eat-term-color-0 ((t :foreground "deep sky blue")))
  (eat-term-color-8 ((t :foreground "grey57")))
  :preface
  (defun nl/open-eat-and-run-command (command)
    "Open a new EAT session and run a shell COMMAND."
    (interactive "sCommand to run: ")
    (let ((buffer-name (generate-new-buffer-name "*eat*"))
          (current-prefix-arg "*eat-command*"))
      (call-interactively 'eat)
      (with-current-buffer buffer-name
        (goto-char (point-max))
        (insert command)
        (comint-send-input))))
  :commands eat-mode)

;; Cache remote-only operations to avoid repeated TRAMP calls
(defmacro my/advice-memoize-remote (fn cache-var)
  "Wrap FN to memoize remote results in CACHE-VAR."
  `(advice-add ,fn :around
               (lambda (orig &rest args)
                 (let ((key (car args)))
                   (if (and key (file-remote-p key))
                       (let ((entry (assoc key ,cache-var)))
                         (if entry
                             (cdr entry)
                           (let ((result (apply orig args)))
                             (setq ,cache-var (cons (cons key result) ,cache-var))
                             result)))
                     (apply orig args))))))

(defvar my/project-current-cache nil)
(my/advice-memoize-remote project-current my/project-current-cache)

(defvar my/magit-toplevel-cache nil)
(my/advice-memoize-remote magit-toplevel my/magit-toplevel-cache)

(defvar my/vc-git-root-cache nil)
(advice-add 'vc-git-root :around
            (lambda (orig file)
              (let ((dir (file-name-directory file)))
                (if (file-remote-p file)
                    (or (cdr (assoc dir my/vc-git-root-cache))
                        (let ((res (funcall orig file)))
                          (setq my/vc-git-root-cache
                                (cons (cons dir res) my/vc-git-root-cache))
                          res))
                  (funcall orig file)))))
