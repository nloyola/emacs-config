** [[https://codeberg.org/akib/emacs-eat][emacs-eat]]

- Keys
~C-c C-c~ :: sends a SIGINT (like pressing =Ctrl+C= in a regular terminal).
~C-c C-d~ :: sends an EOF (like pressing =Ctrl+D=).

#+begin_src emacs-lisp
(use-package eat
  ;; ensure the eat entry points are autoloaded
  :commands (eat eat-other-window)
  :custom-face
  (eat-term-color-0 ((t :foreground "deep sky blue")))
  (eat-term-color-8 ((t :foreground "grey57")))
  :preface
  (defun nl/open-eat-and-run-command (command)
    "Open a new EAT session and run a shell COMMAND."
    (interactive "sCommand to run: ")
    (let ((buffer-name (generate-new-buffer-name "*eat*"))
          (current-prefix-arg "*eat-command*"))
      (call-interactively 'eat)
      (with-current-buffer buffer-name
        (goto-char (point-max))
        (insert command)
        (comint-send-input))))
  :commands eat-mode)
#+end_src

** tramp

#+begin_src emacs-lisp
(use-package tramp
  :ensure nil
  :defer t
  :init
  (setq remote-file-name-inhibit-locks t
        remote-file-name-inhibit-auto-save-visited t
        tramp-copy-size-limit (* 1024 1024)
        tramp-verbose 2)
  :config
  ;; Keep compile using the shared SSH connection for remote commands.
  (with-eval-after-load 'compile
    (remove-hook 'compilation-mode-hook
                 #'tramp-compile-disable-ssh-controlmaster-options)))

(defun nl/bootstrap-tramp-rpc ()
  "Install and load `tramp-rpc' so `/rpc:user@host:/path' paths work.

The Rust server binary is auto-deployed on first connection."
  (interactive)
  (if (version< emacs-version "30.1")
      (message "Skipping tramp-rpc: Emacs 30.1 or newer is required")
    (condition-case err
        (progn
          (require 'tramp)
          (nl/ensure-package-installed 'msgpack)
          (unless (package-installed-p 'tramp-rpc)
            (package-vc-install
             '(tramp-rpc
               :url "https://github.com/ArthurHeymans/emacs-tramp-rpc"
               :lisp-dir "lisp")))
          (require 'tramp-rpc))
      (error
       (message "Failed to enable tramp-rpc: %s"
                (error-message-string err))))))

;; Do not load TRAMP during startup.  If TRAMP is loaded because a remote
;; path is opened, enable the optional /rpc: method then.
(with-eval-after-load 'tramp
  (nl/bootstrap-tramp-rpc))
#+end_src
