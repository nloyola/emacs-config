** [[https://github.com/alphapapa/prism.el][prism]]

#+begin_src emacs-lisp
(use-package prism
  :preface
  (defun nl/prism-colors ()
    "Build a Prism palette from the current theme's font-lock faces."
    (delq nil
          (mapcar (lambda (face)
                    (face-foreground face nil t))
                  '(font-lock-keyword-face
                    font-lock-type-face
                    font-lock-function-name-face
                    font-lock-string-face
                    font-lock-constant-face
                    font-lock-builtin-face
                    font-lock-variable-name-face
                    font-lock-comment-face))))

  (defun nl/configure-prism (&rest _)
    "Refresh Prism faces to match the active theme."
    (when (fboundp 'prism-set-colors)
      (setq prism-num-faces 16)
      (prism-set-colors
       :desaturations '(0)
       :lightens '(0)
       :colors (nl/prism-colors))))
  :hook
  ;; Enable only where indentation/nesting depth carries meaning.
  ;; Skip C-family (PHP, JS/TS, Java, Go, Rust, C/C++) where braces
  ;; delimit scope and prism fights with semantic font-lock faces.
  ;; Paren-delimited languages use `prism-mode'; indentation-based
  ;; languages (Python, YAML, Haskell) use `prism-whitespace-mode',
  ;; which preserves normal font-lock for keywords/strings/etc.
  (((emacs-lisp-mode
     lisp-mode
     lisp-interaction-mode
     clojure-mode
     scheme-mode) . prism-mode)
   ((python-mode
     python-ts-mode
     yaml-mode
     yaml-ts-mode
     haskell-mode) . prism-whitespace-mode))
  :config
  (nl/configure-prism)
  (advice-add 'load-theme :after #'nl/configure-prism))
#+end_src
